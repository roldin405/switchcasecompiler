int choice = 2;
int result = 0;

switch (choice) {
    case 1:
        result = 10;
        print(result);
        break;

    case 2:
        result = 20;
        print(result);
        break;

    case 3:
        result = 30;
        print(result);
        break;

    default:
        result = 0;
        print(result);
        break;
}
